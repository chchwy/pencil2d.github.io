GitHub Pages serves all three forms. */ -}} {{- $u := .Permalink -}} {{- if .Params.url -}} {{- /* an explicit url (404.html) is used as written */ -}} {{- else if ne .Section "posts" -}} {{- $u = $u | strings.TrimSuffix ".html" | strings.TrimSuffix "index" -}} {{- end -}} {{- return $u -}}
