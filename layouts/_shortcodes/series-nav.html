{{- /* Prev / home / next links for a page in a series listed in data/.yml. Titles come from the pages themselves, so a translated page shows its translated title. The links stay site-absolute; _partials/content.html points them at the page's language. */ -}} {{- $name := .Get 0 -}} {{- $series := index hugo.Data $name -}} {{- $here := .Page.RelPermalink | strings.TrimPrefix site.LanguagePrefix | strings.TrimSuffix ".html" | strings.TrimSuffix "index" -}} {{- $items := slice -}} {{- range $series.pages -}} {{- $path := .url | strings.TrimSuffix ".html" | strings.TrimSuffix "/" -}} {{- $title := .title -}} {{- with site.GetPage $path }}{{ $title = .LinkTitle }}{{ end -}} {{- $items = $items | append (dict "url" .url "title" $title) -}} {{- end -}} {{- $idx := -1 -}} {{- range $i, $item := $items -}} {{- if eq (strings.TrimSuffix ".html" $item.url) $here }}{{ $idx = $i }}{{ end -}} {{- end -}} {{- $label := T (printf "series_%s" $name) | default $series.title -}}
{{- if gt $idx 0 -}}{{- $prev := index $items (sub $idx 1) }} ← {{ $prev.title }} {{- else -}}{{- end }} {{ $label }} {{- $n := add $idx 1 -}} {{- if lt $n (len $items) -}}{{- $next := index $items $n }} {{ $next.title }} → {{- else -}}{{- end }}
